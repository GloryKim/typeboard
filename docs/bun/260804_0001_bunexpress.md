# mini — 외부 의존성 0 Express식 웹 프레임워크

`Bun.serve` 위에 얹은 ~250줄짜리 Express 호환 스타일 프레임워크. **런타임 의존성 0** (Bun 표준 API만 사용).

## 파일

| 파일 | 내용 |
|---|---|
| [`mini.ts`](./mini.ts) | 프레임워크 본체 (`Req`/`Res`/`Router`/`App`) |
| [`middleware.ts`](./middleware.ts) | 재사용 미들웨어 (logger·cors·bodyParser·staticDir·requireBearer·errorHandler) |
| [`example.ts`](./example.ts) | 데모 앱 |

## 실행

```bash
bun --hot mini/example.ts
# GET  http://localhost:3000/
# GET  http://localhost:3000/api/users
# POST http://localhost:3000/api/users   { "name": "lin" }
```

## API 개요

```ts
import { mini, router } from "./mini.ts";

const app = mini();

app.use(mw);                       // 전역 미들웨어
app.use("/api", subRouter);        // 서브 라우터 마운트
app.get("/users/:id", handler);    // 메서드 + 경로 파라미터
app.post("/users", mw1, mw2, h);   // 라우트 단위 미들웨어 체인
app.use(errorHandler);             // 4-arg 에러 핸들러

app.listen(3000);
```

### 핸들러 시그니처 (Express 동일)

```ts
(req, res, next) => { ... }              // 일반
(err, req, res, next) => { ... }         // 에러 핸들러 (인자 4개)
```

- `req` — `params` · `query` · `headers` · `cookies` · `body`(파싱 후) · `get(name)` · `path` · `ip`
- `res` — `status()` · `set()` · `json()` · `send()` · `html()` · `redirect()` · `cookie()` · `clearCookie()` (체이닝)
- `next()` — 다음 미들웨어. `next(err)` 로 에러 핸들러로 점프.

## Express 대비 차이

| | Express | mini |
|---|---|---|
| 런타임 | Node + 의존성 다수 | Bun 단독, 의존성 0 |
| 내부 표현 | `http.IncomingMessage` | Web 표준 `Request`/`Response` |
| body 파싱 | `express.json()` 필요 | `bodyParser()` 또는 `req.parseBody()` |
| 스트리밍/WS | 별도 | Bun.serve 네이티브로 확장 가능 |

## 한계 (의도적으로 미구현)

- 뷰 엔진 / 템플릿
- `res.locals`, `app.set/get` 설정 저장소
- 정규식 라우트(`/ab?cd/`) — `:param` 과 `*` 만 지원
- 라우트 `next("route")` 분기


### mini/mini.ts
```ts
// mini.ts — Express식 미니 웹 프레임워크. 외부 의존성 0, Bun.serve만 사용.
//
// 지원: app.use / app.get/post/... / 경로 파라미터(:id) / 와일드카드(*) /
//       미들웨어 체인(next) / 에러 핸들러(4-arg) / res.json/send/status/cookie/redirect /
//       라우터 마운트(app.use("/api", router)) / app.listen()
//
// 설계 메모: 핸들러 시그니처는 Express와 동일한 (req, res, next).
// 응답은 res 의 터미널 메서드(json/send/html/redirect/end)를 호출하면 확정된다.

// ── 타입 ────────────────────────────────────────────────────────────────
export type Method =
  | "GET" | "POST" | "PUT" | "PATCH" | "DELETE" | "HEAD" | "OPTIONS";

export type Next = (err?: unknown) => void;
export type Handler = (req: Req, res: Res, next: Next) => unknown | Promise<unknown>;
export type ErrorHandler =
  (err: unknown, req: Req, res: Res, next: Next) => unknown | Promise<unknown>;

// ── Req ─────────────────────────────────────────────────────────────────
export class Req {
  raw: Request;
  method: string;
  url: URL;
  params: Record<string, string> = {};
  ip?: string;
  private _body: unknown = undefined;
  private _bodyRead = false;

  constructor(raw: Request, ip?: string) {
    this.raw = raw;
    this.method = raw.method;
    this.url = new URL(raw.url);
    this.ip = ip;
  }

  get path() { return this.url.pathname; }
  get query() { return this.url.searchParams; }
  get headers() { return this.raw.headers; }
  get(name: string) { return this.raw.headers.get(name); }

  /** Content-Type 에 맞춰 body 를 한 번만 파싱해 캐시한다. */
  async parseBody(): Promise<unknown> {
    if (this._bodyRead) return this._body;
    this._bodyRead = true;
    const ct = this.raw.headers.get("content-type") ?? "";
    if (this.method === "GET" || this.method === "HEAD") return (this._body = undefined);
    try {
      if (ct.includes("application/json")) this._body = await this.raw.json();
      else if (ct.includes("application/x-www-form-urlencoded") || ct.includes("multipart/form-data"))
        this._body = Object.fromEntries((await this.raw.formData()).entries());
      else this._body = await this.raw.text();
    } catch {
      this._body = undefined;
    }
    return this._body;
  }

  get body() { return this._body; }

  /** 요청 쿠키 (BunRequest 밖에서도 직접 파싱). */
  get cookies() {
    return new Bun.CookieMap(this.raw.headers.get("cookie") ?? "");
  }
}

// ── Res ─────────────────────────────────────────────────────────────────
export class Res {
  statusCode = 200;
  headers = new Headers();
  finished = false;
  private _body: BodyInit | null = null;
  private _cookies: string[] = [];
  private _onEnd: (() => void)[] = [];

  /** 응답이 확정(end)되는 순간 실행할 훅. 로깅 등에 사용. */
  onEnd(fn: () => void): this { this._onEnd.push(fn); return this; }

  status(code: number): this { this.statusCode = code; return this; }
  set(name: string, value: string): this { this.headers.set(name, value); return this; }
  header(name: string, value: string): this { return this.set(name, value); }
  type(contentType: string): this { return this.set("Content-Type", contentType); }

  cookie(name: string, value: string, opts: Bun.CookieInit = {} as Bun.CookieInit): this {
    const c = new Bun.Cookie(name, value, opts);
    this._cookies.push(c.toString());
    return this;
  }
  clearCookie(name: string, opts: Partial<Bun.CookieInit> = {}): this {
    return this.cookie(name, "", { ...opts, maxAge: 0, path: (opts as any).path ?? "/" } as Bun.CookieInit);
  }

  json(body: unknown): this {
    if (!this.headers.has("Content-Type")) this.type("application/json; charset=utf-8");
    this._body = JSON.stringify(body);
    return this.end();
  }
  html(markup: string): this {
    if (!this.headers.has("Content-Type")) this.type("text/html; charset=utf-8");
    this._body = markup;
    return this.end();
  }
  send(body: BodyInit): this {
    if (!this.headers.has("Content-Type") && typeof body === "string")
      this.type("text/plain; charset=utf-8");
    this._body = body;
    return this.end();
  }
  redirect(location: string, code = 302): this {
    this.statusCode = code;
    this.set("Location", location);
    return this.end();
  }
  end(): this {
    if (!this.finished) {
      this.finished = true;
      for (const fn of this._onEnd) fn();
    }
    return this;
  }

  /** 내부: 축적된 상태를 표준 Response 로 변환. */
  toResponse(): Response {
    for (const c of this._cookies) this.headers.append("Set-Cookie", c);
    return new Response(this._body, { status: this.statusCode, headers: this.headers });
  }
}

// ── 경로 컴파일 ───────────────────────────────────────────────────────────
interface Compiled { re: RegExp; keys: string[]; }
function compile(path: string, end: boolean): Compiled {
  const keys: string[] = [];
  // 트레일링 슬래시 제거: "/" → "", "/api/" → "/api".
  // (안 하면 use("/") 가 ^/(?=/|$) 가 되어 "/api/x" 같은 하위 경로에 매칭되지 않는다.)
  const norm = path.replace(/\/+$/, "");
  // 정규식 특수문자 이스케이프 (단 : 와 * 는 문법으로 남긴다)
  let src = norm.replace(/[.+?^${}()|[\]\\]/g, "\\$&");
  src = src.replace(/:([A-Za-z0-9_]+)/g, (_m, k: string) => { keys.push(k); return "([^/]+)"; });
  src = src.replace(/\*/g, () => { keys.push("*"); return "(.*)"; });
  const suffix = end ? "/?$" : "(?=/|$)";
  return { re: new RegExp("^" + src + suffix), keys };
}

function safeDecode(v: string): string {
  try { return decodeURIComponent(v); } catch { return v; }
}

// ── Layer ────────────────────────────────────────────────────────────────
interface Layer {
  method: Method | "ALL";
  compiled: Compiled | null; // null = 모든 경로 (use(mw))
  handler: Handler;
  isError: false;
}
interface ErrLayer { handler: ErrorHandler; isError: true; }

// ── Router / App ──────────────────────────────────────────────────────────
export class Router {
  /** @internal */ stack: (Layer | ErrLayer)[] = [];

  private add(method: Method | "ALL", path: string, handlers: Handler[], end: boolean) {
    const compiled = compile(path, end);
    for (const h of handlers) this.stack.push({ method, compiled, handler: h, isError: false });
    return this;
  }

  use(pathOrHandler: string | Handler | Router | ErrorHandler, ...handlers: (Handler | Router | ErrorHandler)[]): this {
    let path = "/";
    let hs: (Handler | Router | ErrorHandler)[];
    if (typeof pathOrHandler === "string") { path = pathOrHandler; hs = handlers; }
    else hs = [pathOrHandler, ...handlers];

    for (const h of hs) {
      if (h instanceof Router) {
        // 서브 라우터 마운트: prefix 를 붙여 스택을 흡수
        const prefix = path === "/" ? "" : path.replace(/\/$/, "");
        for (const layer of h.stack) {
          if (layer.isError) { this.stack.push(layer); continue; }
          const rawPath = (layer as any)._path ?? "";
          this.stack.push({ ...layer });
          // 컴파일된 정규식을 prefix 반영해 다시 만든다
          const merged = this.stack[this.stack.length - 1] as Layer;
          merged.compiled = compile(prefix + rawPath, (layer as any)._end);
        }
      } else if ((h as Function).length === 4) {
        this.stack.push({ handler: h as ErrorHandler, isError: true });
      } else {
        // use 는 prefix 매칭(end:false)
        const compiled = compile(path, false);
        this.stack.push({ method: "ALL", compiled, handler: h as Handler, isError: false });
      }
    }
    return this;
  }

  // 메서드 라우트 등록. _path/_end 를 붙여 서브라우터 마운트 시 재컴파일에 쓴다.
  private route(method: Method | "ALL", path: string, handlers: Handler[]) {
    for (const h of handlers) {
      const compiled = compile(path, true);
      const layer: Layer & { _path?: string; _end?: boolean } =
        { method, compiled, handler: h, isError: false };
      layer._path = path; layer._end = true;
      this.stack.push(layer);
    }
    return this;
  }

  get(path: string, ...h: Handler[]) { return this.route("GET", path, h); }
  post(path: string, ...h: Handler[]) { return this.route("POST", path, h); }
  put(path: string, ...h: Handler[]) { return this.route("PUT", path, h); }
  patch(path: string, ...h: Handler[]) { return this.route("PATCH", path, h); }
  delete(path: string, ...h: Handler[]) { return this.route("DELETE", path, h); }
  options(path: string, ...h: Handler[]) { return this.route("OPTIONS", path, h); }
  head(path: string, ...h: Handler[]) { return this.route("HEAD", path, h); }
  all(path: string, ...h: Handler[]) { return this.route("ALL", path, h); }
}

export class App extends Router {
  /** Request → Response. Bun.serve 의 fetch 로 그대로 넘길 수 있다. */
  handle = async (raw: Request, server?: Bun.Server): Promise<Response> => {
    const req = new Req(raw, server?.requestIP(raw)?.address);
    const res = new Res();
    const path = req.path;
    const method = req.method as Method;

    // 이 요청에 매칭되는 레이어만 순서대로 수집
    const matched: { layer: Layer; params: Record<string, string> }[] = [];
    const errorLayers: ErrLayer[] = [];
    for (const l of this.stack) {
      if (l.isError) { errorLayers.push(l); continue; }
      const layer = l as Layer;
      if (layer.method !== "ALL" && layer.method !== method) continue;
      const m = layer.compiled!.re.exec(path);
      if (!m) continue;
      const params: Record<string, string> = {};
      layer.compiled!.keys.forEach((k, i) => { params[k] = safeDecode(m[i + 1] ?? ""); });
      matched.push({ layer, params });
    }

    let errored: unknown = undefined;

    const runError = async (i: number, err: unknown): Promise<void> => {
      if (res.finished) return;
      const eh = errorLayers[i];
      if (!eh) { // 처리 안 된 에러 → 500
        res.status(500).type("text/plain; charset=utf-8");
        (res as any)._body = "Internal Server Error";
        res.end();
        return;
      }
      let nextErr: unknown = undefined, nextCalled = false;
      const next: Next = (e) => { nextCalled = true; nextErr = e; };
      try { await eh.handler(err, req, res, next); }
      catch (e) { return runError(i + 1, e); }
      if (res.finished) return;
      if (nextCalled) return runError(i + 1, nextErr !== undefined ? nextErr : err);
    };

    const run = async (i: number): Promise<void> => {
      if (res.finished) return;
      const entry = matched[i];
      if (!entry) return; // 매칭 소진 → 폴스루
      req.params = entry.params;
      let nextErr: unknown = undefined, nextCalled = false;
      const next: Next = (e) => { nextCalled = true; nextErr = e; };
      try { await entry.layer.handler(req, res, next); }
      catch (e) { errored = e; return runError(0, e); }
      if (res.finished) return;
      if (nextCalled) {
        if (nextErr !== undefined) return runError(0, nextErr);
        return run(i + 1);
      }
      // next 도 안 부르고 응답도 안 했으면 여기서 멈춘다(다음 레이어로 안 감).
    };

    await run(0);

    if (!res.finished) {
      // 아무도 응답하지 않음 → 404
      res.status(404).type("text/plain; charset=utf-8");
      (res as any)._body = "Not Found";
      res.end();
    }
    void errored;
    return res.toResponse();
  };

  listen(port = 3000, cb?: (server: Bun.Server) => void): Bun.Server {
    const server = Bun.serve({
      port,
      fetch: (raw, srv) => this.handle(raw, srv),
    });
    cb?.(server);
    return server;
  }
}

/** Express 의 `express()` 처럼 앱 팩토리. */
export function mini(): App { return new App(); }
export function router(): Router { return new Router(); }
```

### mini/middleware.ts
```ts
// middleware.ts — mini 프레임워크용 재사용 미들웨어. 외부 의존성 0.
import type { Handler, ErrorHandler } from "./mini.ts";

/** 요청 로깅: METHOD path → status (ms) */
export const logger = (): Handler => (req, res, next) => {
  const t = performance.now();
  // 응답이 확정되는 순간(end 훅)에 최종 status 로 찍는다.
  res.onEnd(() => {
    const dt = (performance.now() - t).toFixed(1);
    console.log(`${req.method} ${req.path} → ${res.statusCode} (${dt}ms)`);
  });
  next();
};

/** CORS 헤더. preflight(OPTIONS)면 즉시 204 로 끝낸다. */
export const cors = (opts: { origin?: string; methods?: string; headers?: string } = {}): Handler =>
  (req, res, next) => {
    res.set("Access-Control-Allow-Origin", opts.origin ?? "*");
    res.set("Access-Control-Allow-Methods", opts.methods ?? "GET,POST,PUT,PATCH,DELETE,OPTIONS");
    res.set("Access-Control-Allow-Headers", opts.headers ?? "Content-Type, Authorization");
    if (req.method === "OPTIONS") return void res.status(204).send("");
    next();
  };

/** POST/PUT/PATCH 바디를 Content-Type 에 맞춰 파싱해 req.body 에 채운다. */
export const bodyParser = (): Handler => async (req, _res, next) => {
  await req.parseBody();
  next();
};

/** 디렉터리 정적 파일 서버. 경로 이탈(../) 방지 포함. */
export const staticDir = (root: string, mount = "/"): Handler => async (req, res, next) => {
  if (!req.path.startsWith(mount)) return next();
  const rel = decodeURIComponent(req.path.slice(mount.length)) || "index.html";
  if (rel.includes("..") || rel.includes("\0")) return void res.status(400).send("Bad path");
  const file = Bun.file(`${root}/${rel}`);
  if (!(await file.exists())) return next();
  res.status(200);
  (res as any)._body = file;          // BunFile → sendfile 제로카피
  res.type(file.type || "application/octet-stream").end();
};

/** Bearer 토큰 가드. 실패 시 401. */
export const requireBearer = (token: string): Handler => (req, res, next) => {
  const auth = req.get("authorization") ?? "";
  if (auth === `Bearer ${token}`) return next();
  res.status(401).json({ error: "unauthorized" });
};

/** 마지막에 두는 에러 핸들러(4-arg). */
export const errorHandler = (): ErrorHandler => (err, _req, res, next) => {
  if (res.finished) return next(err);
  const status = (err as any)?.status ?? 500;
  const message = (err as any)?.message ?? "Internal Server Error";
  console.error("[mini:error]", err);
  res.status(status).json({ error: message });
};
```

```ts
// example.ts — mini 프레임워크 데모. 실행: bun --hot mini/example.ts
import { mini, router, type Handler } from "./mini.ts";
import { logger, cors, bodyParser, requireBearer, errorHandler } from "./middleware.ts";

const app = mini();

// ── 전역 미들웨어 ──────────────────────────────────────────────
app.use(logger());
app.use(cors());
app.use(bodyParser());

// ── 간단 인메모리 데이터 ────────────────────────────────────────
type User = { id: string; name: string };
const users: User[] = [
  { id: "1", name: "ada" },
  { id: "2", name: "grace" },
];

// ── 라우트 ────────────────────────────────────────────────────
app.get("/", (_req, res) => res.html("<h1>mini</h1><p>Express식, 의존성 0</p>"));

app.get("/hello/:name", (req, res) => {
  res.json({ hello: req.params.name });
});

// 서브 라우터: /api 아래에 마운트
const api = router();
api.get("/users", (_req, res) => res.json(users));
api.get("/users/:id", (req, res) => {
  const u = users.find((x) => x.id === req.params.id);
  if (!u) return void res.status(404).json({ error: "not found" });
  res.json(u);
});
api.post("/users", (req, res) => {
  const body = req.body as { name?: string };
  if (!body?.name) return void res.status(400).json({ error: "name required" });
  const u = { id: crypto.randomUUID(), name: body.name };
  users.push(u);
  res.status(201).json(u);
});

// 보호된 라우트 (미들웨어 여러 개를 라우트 단위로)
const audit: Handler = (req, _res, next) => { console.log("admin hit", req.path); next(); };
api.delete("/users/:id", requireBearer("secret"), audit, (req, res) => {
  const i = users.findIndex((x) => x.id === req.params.id);
  if (i === -1) return void res.status(404).json({ error: "not found" });
  users.splice(i, 1);
  res.status(204).send("");
});

app.use("/api", api);

// 쿠키 세션 데모
app.post("/login", (_req, res) => {
  res.cookie("sid", crypto.randomUUID(), { httpOnly: true, path: "/", maxAge: 3600 });
  res.json({ ok: true });
});
app.get("/me", (req, res) => {
  const sid = req.cookies.get("sid");
  if (!sid) return void res.status(401).json({ error: "no session" });
  res.json({ sid });
});

// 일부러 던지는 에러 → 에러 핸들러로
app.get("/boom", () => { throw Object.assign(new Error("kaboom"), { status: 418 }); });

// 에러 핸들러는 항상 마지막
app.use(errorHandler());

const server = app.listen(3000, (s) => console.log(`mini on ${s.url}`));
export default server;
```
